from contextlib import asynccontextmanager

from fastapi import FastAPI

from api.runtime import AutomationRuntime


runtime = AutomationRuntime("config.json")


@asynccontextmanager
async def lifespan(app: FastAPI):
    runtime.start()

    app.state.runtime = runtime

    try:
        yield
    finally:
        runtime.stop()


app = FastAPI(
    title="Home Automation API",
    version="1.0.0",
    lifespan=lifespan,
)


@app.get("/health")
def get_health():
    error = runtime.get_error()

    return {
        "status": (
            "ok"
            if runtime.is_running()
            else "error"
        ),
        "hardware_loop_running": runtime.is_running(),
        "error": (
            str(error)
            if error is not None
            else None
        ),
    }
