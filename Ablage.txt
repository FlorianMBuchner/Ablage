import time
from smbus2 import SMBus


class PCA9685:
    MODE1 = 0x00
    MODE2 = 0x01
    PRESCALE = 0xFE
    LED0_ON_L = 0x06

    def __init__(self, bus_number=1, address=0x40):
        self.address = address
        self.bus = SMBus(bus_number)

        # Auto-Increment aktivieren
        self.bus.write_byte_data(self.address, self.MODE1, 0x20)

        # Totem-Pole-Ausgänge
        self.bus.write_byte_data(self.address, self.MODE2, 0x04)

        time.sleep(0.01)

    def set_frequency(self, frequency):
        prescale = round(25_000_000 / (4096 * frequency)) - 1
        prescale = max(3, min(255, prescale))

        old_mode = self.bus.read_byte_data(self.address, self.MODE1)
        sleep_mode = (old_mode & 0x7F) | 0x10

        self.bus.write_byte_data(self.address, self.MODE1, sleep_mode)
        self.bus.write_byte_data(self.address, self.PRESCALE, prescale)
        self.bus.write_byte_data(self.address, self.MODE1, old_mode)

        time.sleep(0.005)

        # Neustart und Auto-Increment
        self.bus.write_byte_data(
            self.address,
            self.MODE1,
            old_mode | 0xA0
        )

    def set_pwm(self, channel, on, off):
        if not 0 <= channel <= 15:
            raise ValueError("Kanal muss zwischen 0 und 15 liegen")

        register = self.LED0_ON_L + 4 * channel

        self.bus.write_i2c_block_data(
            self.address,
            register,
            [
                on & 0xFF,
                (on >> 8) & 0x0F,
                off & 0xFF,
                (off >> 8) & 0x0F,
            ],
        )

    def set_servo_us(self, channel, pulse_us, frequency=50):
        period_us = 1_000_000 / frequency
        ticks = round(pulse_us * 4096 / period_us)
        ticks = max(0, min(4095, ticks))

        self.set_pwm(channel, 0, ticks)

    def close(self):
        self.bus.close()


pca = PCA9685(address=0x40)
pca.set_frequency(50)

try:
    # Servo an Kanal 0 bewegen
    pca.set_servo_us(0, 1000)
    time.sleep(1)

    pca.set_servo_us(0, 1500)
    time.sleep(1)

    pca.set_servo_us(0, 2000)
    time.sleep(1)
finally:
    pca.close()
